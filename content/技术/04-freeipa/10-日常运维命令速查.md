# 10 — 日常运维命令速查

## 10.1 容器操作

```bash
# 容器管理
docker start freeipa-server          # 启动容器
docker stop freeipa-server           # 停止容器
docker restart freeipa-server        # 重启容器
docker logs -f freeipa-server        # 查看日志（实时）
docker logs --tail 100 freeipa-server # 查看最近 100 行日志
docker exec -it freeipa-server bash  # 进入容器 Shell

# 状态检查
docker ps | grep freeipa             # 容器是否运行
docker stats freeipa-server          # 资源使用情况

# 容器内 IPA 服务管理
docker exec freeipa-server ipactl status   # 查看服务状态
docker exec freeipa-server ipactl start    # 启动所有服务
docker exec freeipa-server ipactl stop     # 停止所有服务
docker exec freeipa-server ipactl restart  # 重启所有服务
```

## 10.2 Kerberos 认证

```bash
# 获取票据（进入容器后或客户端执行）
echo "PASSWORD" | kinit admin        # admin 认证
kinit zhangsan                        # 普通用户认证（交互输入密码）
klist                                 # 查看当前票据
klist -A                              # 查看所有票据详情
kdestroy                              # 销毁当前票据
kinit -R                              # 续期票据

# 修改密码
kpasswd                               # 修改自己的 Kerberos 密码
kpasswd zhangsan                      # 修改他人密码（需 admin 权限）
ipa passwd zhangsan                   # IPA 方式修改密码

# 密码策略相关
kadmin.local -q "getprinc admin"      # 查看 principal 详情（需在 IPA 服务器上）
```

## 10.12 常用运维脚本模板

### 每日巡检脚本

```bash
#!/bin/bash
# daily-check.sh — 每日巡检

echo "=== FreeIPA 每日巡检 - $(date) ==="
echo ""

# 1. 容器状态
if docker ps | grep -q freeipa-server; then
  echo "✓ 容器运行中"
else
  echo "✗ 容器未运行！"
fi

# 2. IPA 服务
SERVICES=$(docker exec freeipa-server ipactl status 2>/dev/null | grep -c "RUNNING")
if [ "$SERVICES" -ge 5 ]; then
  echo "✓ IPA 核心服务正常"
else
  echo "✗ IPA 服务异常 ($SERVICES 个运行中)"
fi

# 3. Kerberos
if docker exec freeipa-server bash -c "echo 'YourAdminPassword' | kinit admin" 2>/dev/null; then
  echo "✓ Kerberos 认证正常"
else
  echo "✗ Kerberos 认证失败"
fi

# 4. 磁盘空间
DISK_USAGE=$(df -h /data/docker-space/ | tail -1 | awk '{print $5}' | tr -d '%')
echo "  数据目录磁盘使用: ${DISK_USAGE}%"

# 5. 即将过期证书
EXPIRING=$(docker exec freeipa-server ipa cert-find --validnotafter-to="$(date -u -d '+30 days' +%Y-%m-%d)" 2>/dev/null | grep "Serial number" | wc -l)
echo "  30天内过期证书: $EXPIRING 张"

echo ""
echo "=== 巡检结束 ==="
```

### 批量用户创建模板

```bash
#!/bin/bash
# bulk-user-create.sh

# CSV 格式：username,firstname,lastname,password,department
while IFS=',' read -r username first last password dept; do
  [[ "$username" =~ ^#.* ]] && continue  # 跳过注释行
  [[ -z "$username" ]] && continue       # 跳过空行
  
  echo "创建: $username ($last$first)"
  ipa user-add "$username" \
    --first="$first" \
    --last="$last" \
    --password="$password"
  
  ipa group-add-member "$dept" --users="$username" 2>/dev/null
done < users.csv
```

### 批量密码重置

```bash
#!/bin/bash
# bulk-password-reset.sh

USERS=("zhangsan" "lisi" "wangwu")
NEW_PASSWORD="TempPass123"

for user in "${USERS[@]}"; do
  echo "重置密码: $user"
  echo -e "${NEW_PASSWORD}\n${NEW_PASSWORD}" | ipa passwd "$user"
  # 强制下次登录修改密码
  ipa user-mod "$user" --setattr=krbPasswordExpiration="$(date -u +%Y%m%d%H%M%SZ -d '+1 day')"
done
```
